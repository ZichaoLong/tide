cmake_minimum_required(VERSION 3.20)
project(tide_external_llama LANGUAGES C CXX)
if(NOT DEFINED LLAMA_SOURCE)
  message(FATAL_ERROR "Pass -DLLAMA_SOURCE=/absolute/pinned/llama.cpp")
endif()
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP ON CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(tide-lm-llama main.cpp)
target_link_libraries(tide-lm-llama PRIVATE llama)
