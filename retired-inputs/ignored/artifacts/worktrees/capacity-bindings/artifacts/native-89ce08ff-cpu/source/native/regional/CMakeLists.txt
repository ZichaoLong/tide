cmake_minimum_required(VERSION 3.18)
project(tide_regional LANGUAGES CXX)
set(TIDE_NATIVE_BACKEND "CPU" CACHE STRING "CPU or NPU standalone binary")
set_property(CACHE TIDE_NATIVE_BACKEND PROPERTY STRINGS CPU NPU)
if(NOT TIDE_NATIVE_BACKEND MATCHES "^(CPU|NPU)$")
  message(FATAL_ERROR "TIDE_NATIVE_BACKEND must be CPU or NPU")
endif()
find_package(Torch CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(tide-regional main.cpp region.cpp bundle.cpp runtime.cpp)
target_compile_features(tide-regional PRIVATE cxx_std_17)
target_link_libraries(tide-regional PRIVATE torch Threads::Threads)
separate_arguments(TIDE_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(tide-regional PRIVATE ${TIDE_TORCH_FLAGS})
if(TIDE_NATIVE_BACKEND STREQUAL "NPU")
  find_package(Torch_npu CONFIG REQUIRED)
  if(NOT TORCH_NPU_INCLUDE_DIRS OR NOT TORCH_NPU_LIBRARIES)
    message(FATAL_ERROR "The standalone SDK must export public include/library metadata")
  endif()
  target_include_directories(tide-regional PRIVATE ${TORCH_NPU_INCLUDE_DIRS})
  target_link_libraries(tide-regional PRIVATE ${TORCH_NPU_LIBRARIES})
  target_compile_definitions(tide-regional PRIVATE TIDE_WITH_NPU=1)
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                OUTPUT_VARIABLE TIDE_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git status --porcelain --untracked-files=all
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE TIDE_SOURCE_STATUS)
if(TIDE_SOURCE_STATUS)
  set(TIDE_SOURCE_DIRTY true)
else()
  set(TIDE_SOURCE_DIRTY false)
endif()
configure_file(build_identity.h.in build_identity.h @ONLY)
target_include_directories(tide-regional PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME regional_help COMMAND tide-regional --help)
add_test(NAME regional_reject_unknown COMMAND tide-regional --unknown)
set_tests_properties(regional_reject_unknown PROPERTIES WILL_FAIL TRUE)
add_test(NAME regional_reject_implicit_device COMMAND tide-regional)
set_tests_properties(regional_reject_implicit_device PROPERTIES WILL_FAIL TRUE)
