cmake_minimum_required(VERSION 3.18)
project(tide_lh_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Torch REQUIRED)
find_package(OpenMP REQUIRED)
if(NOT TIDE_LH_SNAPSHOT)
  message(FATAL_ERROR "explicit frozen LH source snapshot required")
endif()
set(LH_SOURCES BaseConfig GraphConfig Adjacency Selector AccumulateLocal Hidden
  BatchHidden Confluence ModuleUtils CortexConfig CortexNet)
foreach(name IN LISTS LH_SOURCES)
  list(APPEND LH_FILES ${TIDE_LH_SNAPSHOT}/src/${name}.cpp)
endforeach()
add_library(original_lh_bench STATIC ${LH_FILES})
target_include_directories(original_lh_bench PUBLIC ${TIDE_LH_SNAPSHOT}/include ${TIDE_LH_SNAPSHOT}/vendor)
target_compile_definitions(original_lh_bench PUBLIC DISABLE_CUDA ENABLE_PARALLEL_FOR)
target_link_libraries(original_lh_bench PUBLIC torch OpenMP::OpenMP_CXX)
separate_arguments(TIDE_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(original_lh_bench PUBLIC ${TIDE_TORCH_FLAGS})
add_executable(tide-lh-bench main.cpp)
target_link_libraries(tide-lh-bench PRIVATE original_lh_bench)
