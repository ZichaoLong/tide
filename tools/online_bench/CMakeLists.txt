cmake_minimum_required(VERSION 3.18)
project(tide_online_consumer LANGUAGES CXX)
find_package(TideGraph CONFIG REQUIRED)
add_executable(tidegraph-online-bench main.cpp config.cpp packet.cpp model.cpp run.cpp records.cpp memory.cpp)
target_link_libraries(tidegraph-online-bench PRIVATE tide::runtime ${CMAKE_DL_LIBS})
target_compile_features(tidegraph-online-bench PRIVATE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
  add_executable(tidegraph-eager-capacity-probe ../../tests/eager_capacity_probe.cpp config.cpp packet.cpp records.cpp)
  target_include_directories(tidegraph-eager-capacity-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(tidegraph-eager-capacity-probe PRIVATE tide::runtime)
  target_compile_features(tidegraph-eager-capacity-probe PRIVATE cxx_std_17)
  add_executable(tidegraph-eager-precision-probe ../../tests/eager_precision_probe.cpp records.cpp)
  target_include_directories(tidegraph-eager-precision-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(tidegraph-eager-precision-probe PRIVATE tide::runtime)
  target_compile_features(tidegraph-eager-precision-probe PRIVATE cxx_std_17)
endif()
option(TIDE_ONLINE_RESIDENT "Build the optional public CANN resident consumer" OFF)
if(TIDE_ONLINE_RESIDENT)
  find_package(TideResident CONFIG REQUIRED)
  target_sources(tidegraph-online-bench PRIVATE resident_run.cpp resident_loss.cpp resident_records.cpp resident_capacity.cpp)
  target_link_libraries(tidegraph-online-bench PRIVATE tide::resident)
  target_compile_definitions(tidegraph-online-bench PRIVATE TIDE_ONLINE_RESIDENT=1)
endif()
