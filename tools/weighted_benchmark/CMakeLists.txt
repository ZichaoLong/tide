cmake_minimum_required(VERSION 3.18)
project(tide_weighted_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TIDE_WEIGHTED_RESIDENT "Build the standalone resident consumer too" OFF)
find_package(TideGraph CONFIG REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_path(TIDE_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_executable(weighted-benchmark main.cpp cli.cpp run_checkpoint.cpp validation.cpp roots.cpp root_values.cpp eager_roots.cpp eager.cpp fixture.cpp tensors.cpp records.cpp return_packing.cpp record_io.cpp observation.cpp allocation_failure.cpp ../../cpp/bench/compare.cpp)
target_sources(weighted-benchmark PRIVATE ../device_online/weighted_precision_reference.cpp)
target_include_directories(weighted-benchmark PRIVATE "${TIDE_JSON_INCLUDE_DIR}")
target_link_libraries(weighted-benchmark PRIVATE tide::runtime OpenSSL::Crypto)
install(PROGRAMS "$<TARGET_FILE:weighted-benchmark>" DESTINATION bin RENAME tidegraph-core5-run)
add_executable(weighted-return-check return_packing_check.cpp return_packing.cpp)
target_link_libraries(weighted-return-check PRIVATE tide::runtime)
add_executable(weighted-comparison-check comparison_check.cpp ../../cpp/bench/compare.cpp)
target_link_libraries(weighted-comparison-check PRIVATE tide::runtime)
add_executable(weighted-run-check run_check.cpp cli.cpp run_checkpoint.cpp record_io.cpp)
target_include_directories(weighted-run-check PRIVATE "${TIDE_JSON_INCLUDE_DIR}")
target_link_libraries(weighted-run-check PRIVATE tide::runtime)
add_executable(weighted-roots-check roots_check.cpp root_values.cpp ../../cpp/bench/compare.cpp)
target_link_libraries(weighted-roots-check PRIVATE tide::runtime)
enable_testing()
add_test(NAME weighted-comparison-check COMMAND weighted-comparison-check)
add_test(NAME weighted-run-check COMMAND weighted-run-check)
add_test(NAME weighted-roots-check COMMAND weighted-roots-check)
if(TIDE_WEIGHTED_RESIDENT)
  find_package(TideWeightedResident CONFIG REQUIRED COMPONENTS training)
  target_sources(weighted-benchmark PRIVATE resident.cpp resident_checkpoint.cpp)
  target_compile_definitions(weighted-benchmark PRIVATE TIDE_WEIGHTED_RESIDENT=1)
  target_link_libraries(weighted-benchmark PRIVATE tide::weighted_training)
endif()
