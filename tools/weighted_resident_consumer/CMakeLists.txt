cmake_minimum_required(VERSION 3.18)
project(tide_weighted_resident_consumer LANGUAGES CXX)
option(TIDE_WEIGHTED_TRAINING "Check the installed explicit training API" OFF)
if(TIDE_WEIGHTED_TRAINING)
  find_package(TideWeightedResident CONFIG REQUIRED COMPONENTS training)
  add_executable(weighted-training-consumer training.cpp)
  target_link_libraries(weighted-training-consumer PRIVATE tide::weighted_training tide::runtime)
else()
  find_package(TideWeightedResident CONFIG REQUIRED)
  add_executable(weighted-resident-consumer main.cpp)
  target_link_libraries(weighted-resident-consumer PRIVATE tide::weighted_resident tide::runtime)
endif()
