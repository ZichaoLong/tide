cmake_minimum_required(VERSION 3.18)
project(tide_accelerator_scale LANGUAGES CXX)
find_package(TideGraph 0.2 CONFIG REQUIRED)
add_library(tide-scale-support STATIC transfer.cpp partition.cpp placement.cpp state.cpp full.cpp resident.cpp scoring.cpp selection.cpp event_queue.cpp
  ../../cpp/scale/config.cpp ../../cpp/scale/model.cpp ../../cpp/bench/compare.cpp)
target_link_libraries(tide-scale-support PUBLIC tide::runtime)
target_compile_features(tide-scale-support PUBLIC cxx_std_17)
add_executable(tide-accelerator-scale main.cpp check.cpp check_vjp.cpp training.cpp training_check.cpp precision.cpp profiling.cpp)
target_link_libraries(tide-accelerator-scale PRIVATE tide-scale-support ${CMAKE_DL_LIBS})
target_compile_features(tide-accelerator-scale PRIVATE cxx_std_17)
add_library(tide-bounded-scheduler STATIC bounded_program.cpp bounded_update.cpp bounded_select.cpp bounded_export.cpp bounded_vjp.cpp bounded_optimizer.cpp graph_replay.cpp peer_transport.cpp)
target_link_libraries(tide-bounded-scheduler PUBLIC tide-scale-support ${CMAKE_DL_LIBS})
if(TORCH_NPU_INCLUDE_DIRS)
  target_include_directories(tide-bounded-scheduler PRIVATE "${TORCH_NPU_INCLUDE_DIRS}/third_party/acl/inc")
endif()
add_executable(tide-bounded-schedule-check bounded_check.cpp bounded_train_check.cpp bounded_replay_check.cpp bounded_peer_check.cpp precision.cpp)
target_link_libraries(tide-bounded-schedule-check PRIVATE tide-bounded-scheduler ${CMAKE_DL_LIBS})
add_executable(tide-bounded-scale bounded_benchmark.cpp)
target_link_libraries(tide-bounded-scale PRIVATE tide-bounded-scheduler)
add_executable(tide-complete-flow flow_main.cpp flow_config.cpp flow_fixture.cpp flow_execution.cpp
  flow_check.cpp flow_roots.cpp flow_poison.cpp flow_benchmark.cpp precision.cpp profiling.cpp)
target_link_libraries(tide-complete-flow PRIVATE tide-bounded-scheduler ${CMAKE_DL_LIBS})
enable_testing()

target_link_libraries(tide-accelerator-vjp-test PRIVATE tide::runtime)

add_test(NAME accelerator-vjp-policy COMMAND tide-accelerator-vjp-test)

target_link_libraries(tide-accelerator-scoring-test PRIVATE tide::runtime)
target_compile_features(tide-accelerator-scoring-test PRIVATE cxx_std_17)
add_test(NAME accelerator-scoring COMMAND tide-accelerator-scoring-test)

target_link_libraries(tide-accelerator-dispatch-test PRIVATE tide::runtime)
target_compile_features(tide-accelerator-dispatch-test PRIVATE cxx_std_17)
add_test(NAME accelerator-dispatch COMMAND tide-accelerator-dispatch-test cpu)
add_executable(tide-accelerator-precision-test precision_test.cpp precision.cpp)
target_link_libraries(tide-accelerator-precision-test PRIVATE tide::runtime ${CMAKE_DL_LIBS})
