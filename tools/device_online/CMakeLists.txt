cmake_minimum_required(VERSION 3.18)
project(tide_device_online LANGUAGES CXX)
option(TIDE_DEVICE_ASCENDC "Build explicit Ascend C queue closure" OFF)
find_package(TideGraph 0.2 CONFIG REQUIRED)
if(NOT TARGET TideGraph::NpuSDK)
  message(FATAL_ERROR "Device control requires a standalone NPU TideGraph package")
endif()
add_library(tide-device-control STATIC cann_api.cpp cann_program.cpp)
target_link_libraries(tide-device-control PUBLIC tide::runtime ${CMAKE_DL_LIBS})
target_compile_features(tide-device-control PUBLIC cxx_std_17)
set_target_properties(tide-device-control PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(tide-device-control-check control_check.cpp)
target_link_libraries(tide-device-control-check PRIVATE tide-device-control)
add_executable(tide-device-numerical-check numerical_check.cpp)
target_link_libraries(tide-device-numerical-check PRIVATE tide-device-control)
add_library(tide-packed-queue STATIC packed_queue.cpp queue_closure.cpp)
target_link_libraries(tide-packed-queue PUBLIC tide::runtime)
target_compile_features(tide-packed-queue PUBLIC cxx_std_17)
set_target_properties(tide-packed-queue PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(tide-packed-queue-check queue_check.cpp)
target_link_libraries(tide-packed-queue-check PRIVATE tide-packed-queue)
enable_testing()
add_test(NAME device-control-help COMMAND tide-device-control-check --help)
add_test(NAME device-numerical-help COMMAND tide-device-numerical-check --help)
add_test(NAME packed-queue-cpu-fp32 COMMAND tide-packed-queue-check --device=cpu --dtype=float32)
add_test(NAME packed-queue-cpu-fp64 COMMAND tide-packed-queue-check --device=cpu --dtype=float64)
if(TIDE_DEVICE_ASCENDC)
  add_subdirectory(ascendc)
  add_executable(tide-device-closure-check closure_check.cpp)
  target_compile_options(tide-device-closure-check PRIVATE -fPIC)
  target_link_libraries(tide-device-closure-check PRIVATE tide-device-control tide-packed-queue tide_closure_kernel)
  target_compile_definitions(tide-device-closure-check PRIVATE TIDE_ASCENDC_SOC="${SOC_VERSION}")
  # CANN's generated kernel library has a runtime-registration constructor;
  # even its --help belongs in a device-leased gate, not a CPU build CTest.
endif()
