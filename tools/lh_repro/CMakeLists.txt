cmake_minimum_required(VERSION 3.18)
project(lh_repro_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Torch CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
set(LH_SOURCES BaseConfig GraphConfig CortexConfig ModuleUtils AccumulateLocal
  Hidden Adjacency CortexNet Selector Confluence BatchHidden)
foreach(name IN LISTS LH_SOURCES)
  list(APPEND LH_FILES src/${name}.cpp)
endforeach()
add_library(Connectome SHARED ${LH_FILES})
target_include_directories(Connectome PUBLIC include repro/vendor)
target_compile_definitions(Connectome PUBLIC DISABLE_CUDA ENABLE_PARALLEL_FOR ENABLE_RAIITIMER)
separate_arguments(LH_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(Connectome PUBLIC ${LH_TORCH_FLAGS})
target_link_libraries(Connectome PUBLIC torch OpenMP::OpenMP_CXX)
foreach(mode IN ITEMS grad nograd)
  add_executable(test-cortexnet-${mode} test/test-cortexnet.cpp)
  add_executable(test-cortexnet-info-${mode} test/test-cortexnet-info.cpp)
  target_link_libraries(test-cortexnet-${mode} PRIVATE Connectome)
  target_link_libraries(test-cortexnet-info-${mode} PRIVATE Connectome)
endforeach()
target_compile_definitions(test-cortexnet-nograd PRIVATE NOGRAD)
target_compile_definitions(test-cortexnet-info-nograd PRIVATE NOGRAD)
message(STATUS "LH reproduction: CPU, original node parallelism, original Think timers")
message(STATUS "Compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}; system: ${CMAKE_SYSTEM_PROCESSOR}")
message(STATUS "Torch version: ${Torch_VERSION}; prefix: ${CMAKE_PREFIX_PATH}; build type: ${CMAKE_BUILD_TYPE}")
