cmake_minimum_required(VERSION 3.18)
project(tidegraph VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Torch REQUIRED)
find_package(Threads REQUIRED)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(TIDE_BUILD_CLIENTS "Build repository checks and benchmark clients" ON)
option(TIDE_PYTHON_BINDINGS "Build the Python test/client adapter" ON)
include(CTest)
set(TIDE_BACKEND "CPU" CACHE STRING "Native accelerator family: CPU, CUDA or NPU")
set_property(CACHE TIDE_BACKEND PROPERTY STRINGS CPU CUDA NPU)
set(TIDE_NPU_RUNTIME "standalone" CACHE STRING "NPU registration owner: standalone or python")
if(NOT TIDE_BACKEND MATCHES "^(CPU|CUDA|NPU)$")
  message(FATAL_ERROR "TIDE_BACKEND must be CPU, CUDA or NPU")
endif()
if(TIDE_BACKEND STREQUAL "CUDA" AND NOT TARGET torch_cuda)
  message(FATAL_ERROR "CUDA requires a matching CUDA LibTorch distribution")
endif()
if(TIDE_BACKEND STREQUAL "NPU")
  if(TIDE_NPU_RUNTIME STREQUAL "standalone")
    if(TIDE_PYTHON_BINDINGS)
      message(FATAL_ERROR "Standalone libtorch_npu must not share a process with the Python wheel; set TIDE_PYTHON_BINDINGS=OFF or TIDE_NPU_RUNTIME=python")
    endif()
    include(cmake/NpuSDK.cmake)
  elseif(TIDE_NPU_RUNTIME STREQUAL "python")
    if(TIDE_BUILD_CLIENTS OR NOT TIDE_PYTHON_BINDINGS)
      message(FATAL_ERROR "Python-owned NPU registration requires bindings ON and standalone clients OFF")
    endif()
  else()
    message(FATAL_ERROR "TIDE_NPU_RUNTIME must be standalone or python")
  endif()
endif()
add_library(tidegraph STATIC
  cpp/src/device.cpp
  cpp/src/graph.cpp cpp/src/ports.cpp cpp/src/ops.cpp cpp/src/full_kernel.cpp cpp/src/full_evaluate.cpp cpp/src/isolated_linear.cpp cpp/src/full_rows.cpp cpp/src/row_emit.cpp cpp/src/pool.cpp cpp/src/dense.cpp
  cpp/src/aggregate_kernel.cpp cpp/src/aggregate_evaluate.cpp cpp/src/isolated_aggregate.cpp cpp/src/source_domain.cpp cpp/src/read.cpp cpp/src/next.cpp
  cpp/src/region.cpp cpp/src/region_evaluate.cpp cpp/src/lh_selector.cpp cpp/src/lazy_add.cpp cpp/src/lh_full.cpp
  cpp/src/placement.cpp cpp/src/placement_read.cpp cpp/src/placement_region.cpp
  cpp/src/fiber_attention.cpp cpp/src/fiber_packing.cpp cpp/src/fiber_single.cpp cpp/src/fiber_rows.cpp
  cpp/src/fiber_pool.cpp cpp/src/fiber_pool_csr.cpp cpp/src/token_window.cpp
  cpp/src/validation.cpp cpp/src/input_validation.cpp cpp/src/stream.cpp cpp/src/stream_support.cpp cpp/src/cursor.cpp cpp/src/planner.cpp
  cpp/src/block.cpp cpp/src/block_step.cpp cpp/src/frontier.cpp cpp/src/greedy.cpp cpp/src/greedy_plan.cpp cpp/src/specialized.cpp cpp/src/settle.cpp cpp/src/settle_projection.cpp
  cpp/src/kernel.cpp cpp/src/state_evaluate.cpp cpp/src/clock.cpp cpp/src/clocked_kernel.cpp cpp/src/autograd.cpp cpp/src/basic_kernels.cpp cpp/src/matrix_kernels.cpp
  cpp/src/packed.cpp cpp/src/attention.cpp cpp/src/block_prepare.cpp cpp/src/parameters.cpp cpp/src/optimizer.cpp
  cpp/src/checkpoint.cpp cpp/src/checkpoint_codec.cpp cpp/src/checkpoint_preflight.cpp cpp/src/checkpoint_io.cpp)
set_target_properties(tidegraph PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(tide::tidegraph ALIAS tidegraph)
target_include_directories(tidegraph PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cpp/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(tidegraph PUBLIC cxx_std_17)
target_link_libraries(tidegraph PUBLIC torch Threads::Threads)
target_compile_definitions(tidegraph PUBLIC
  PORTABLE_TORCH_ENABLE_CUDA=$<STREQUAL:${TIDE_BACKEND},CUDA>
  PORTABLE_TORCH_ENABLE_NPU=$<STREQUAL:${TIDE_BACKEND},NPU>)
separate_arguments(TIDE_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(tidegraph PUBLIC ${TIDE_TORCH_FLAGS})
add_library(tidegraph_runtime STATIC cpp/src/runtime.cpp cpp/src/runtime_cli.cpp)
set_target_properties(tidegraph_runtime PROPERTIES EXPORT_NAME runtime POSITION_INDEPENDENT_CODE ON)
add_library(tide::runtime ALIAS tidegraph_runtime)
target_link_libraries(tidegraph_runtime PUBLIC tidegraph)
if(TIDE_BACKEND STREQUAL "NPU" AND TIDE_NPU_RUNTIME STREQUAL "standalone")
  target_compile_definitions(tidegraph_runtime PRIVATE TIDE_NPU_STANDALONE=1)
  target_link_libraries(tidegraph_runtime PUBLIC TideGraph::NpuSDK)
endif()

target_link_libraries(tidegraph-accelerator-check PRIVATE tidegraph_runtime)

target_include_directories(tidegraph-profile-check PRIVATE cpp/include)
target_link_libraries(tidegraph-profile-check PRIVATE Threads::Threads)

target_link_libraries(tidegraph-settle-check PRIVATE tidegraph_runtime)

add_executable(tidegraph-placement-check cpp/test/placement.cpp cpp/bench/compare.cpp)
target_link_libraries(tidegraph-placement-check PRIVATE tidegraph_runtime)
if(BUILD_TESTING AND TIDE_BACKEND STREQUAL "CPU")
  foreach(dtype float32 float64)
    add_test(NAME runtime-${dtype} COMMAND tidegraph-runtime-check --device=cpu --dtype=${dtype})
    add_test(NAME greedy-${dtype} COMMAND tidegraph-greedy-check --device=cpu --dtype=${dtype})
    add_test(NAME settle-${dtype} COMMAND tidegraph-settle-check --device=cpu --dtype=${dtype})
    add_test(NAME read-precision-${dtype} COMMAND tidegraph-read-precision-check --device=cpu --dtype=${dtype})
    add_test(NAME placement-${dtype} COMMAND tidegraph-placement-check --device=cpu --dtype=${dtype})
  endforeach()

add_executable(tidegraph-kernel-check cpp/test/custom_kernel.cpp cpp/test/content_programs.cpp cpp/test/read_programs.cpp cpp/test/next_programs.cpp cpp/test/region_programs.cpp cpp/test/fiber_policy.cpp cpp/test/state_clocks.cpp)

target_link_libraries(tidegraph-checkpoint-check PRIVATE tidegraph_runtime)

  cpp/bench/workload.cpp cpp/bench/compare.cpp)

endif()
if(TIDE_PYTHON_BINDINGS)
  find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
  find_library(TORCH_PYTHON_LIBRARY torch_python PATHS "${TORCH_INSTALL_PREFIX}/lib" REQUIRED NO_DEFAULT_PATH)
  Python3_add_library(_tide_native MODULE cpp/src/bindings.cpp cpp/src/bind_settle.cpp cpp/src/bind_metrics.cpp cpp/src/bind_full.cpp cpp/src/bind_aggregate.cpp cpp/src/bind_placement.cpp)
  target_link_libraries(_tide_native PRIVATE tidegraph ${TORCH_PYTHON_LIBRARY})

install(TARGETS tidegraph tidegraph_runtime EXPORT TideGraphTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY cpp/include/tide DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY cpp/include/portable_torch DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT TideGraphTargets FILE TideGraphTargets.cmake NAMESPACE tide::
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TideGraph)
configure_package_config_file(cmake/TideGraphConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/TideGraphConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TideGraph)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/TideGraphConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/TideGraphConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/TideGraphConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TideGraph)
install(FILES cmake/NpuSDK.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TideGraph)
