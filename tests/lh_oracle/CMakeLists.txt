cmake_minimum_required(VERSION 3.18)
project(tide_lh_selector_oracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Torch REQUIRED)
find_package(Threads REQUIRED)
if(NOT TIDE_LH_SNAPSHOT OR NOT TIDE_CORE_LIBRARY)
  message(FATAL_ERROR "explicit immutable LH snapshot and qualified Tide core library are required")
endif()
get_filename_component(TIDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(original_lh STATIC
  ${TIDE_LH_SNAPSHOT}/src/BaseConfig.cpp ${TIDE_LH_SNAPSHOT}/src/GraphConfig.cpp
  ${TIDE_LH_SNAPSHOT}/src/Adjacency.cpp ${TIDE_LH_SNAPSHOT}/src/Selector.cpp)
target_include_directories(original_lh PUBLIC ${TIDE_LH_SNAPSHOT}/include ${TIDE_LH_SNAPSHOT}/vendor)
target_compile_definitions(original_lh PUBLIC DISABLE_CUDA ENABLE_RUNTIME_ASSERTION)
target_link_libraries(original_lh PUBLIC torch)
separate_arguments(TIDE_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(original_lh PUBLIC ${TIDE_TORCH_FLAGS} -UNDEBUG)
add_library(qualified_tide STATIC IMPORTED)
set_target_properties(qualified_tide PROPERTIES IMPORTED_LOCATION "${TIDE_CORE_LIBRARY}")
add_executable(lh-selector-check ${TIDE_ROOT}/cpp/test/lh_selector_oracle.cpp ${TIDE_ROOT}/cpp/src/runtime.cpp)
target_include_directories(lh-selector-check PRIVATE ${TIDE_ROOT}/cpp/include)
target_link_libraries(lh-selector-check PRIVATE qualified_tide original_lh Threads::Threads)
